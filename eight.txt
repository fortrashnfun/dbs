-- Query 2
SELECT M.*
FROM MATCHES M, BATTING B, PLAYER P
WHERE M.MatchId = B.MatchId
AND B.PId = P.PId
AND P.Lname = 'Dhoni';

FROM PLAYER P, BATTING B
WHERE P.Pid = B.Pid
AND MatchId = 2689;
