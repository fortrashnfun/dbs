-- Query 1
SELECT Dname, AVG(Salary)
FROM EMPLOYEE E, DEPARTMENT D
WHERE E.Dno = D.Dnumber
GROUP BY Dname;

-- Query 2
SELECT Fname, Lname
FROM EMPLOYEE E, DEPARTMENT D, DEPENDENT DP
WHERE E.SSN = D.MgrSSN
AND E.SSN = DP.ESSN;

-- Query 3
SELECT *
FROM DEPARTMENT
WHERE Dname LIKE '%tech%';
