//SQL

-- i. Find the names of sailors who have reserved at least one boat

SELECT DISTINCT Sname
FROM SAILORS S, RESERVES R
WHERE S.Sid = R.Sid;

-- ii. Find the Sid’s of sailors who have reserved
-- a red or a green boat

SELECT DISTINCT S.Sid
FROM SAILORS S, RESERVES R, BOATS B
WHERE S.Sid = R.Sid
AND R.Bid = B.Bid
AND Colour IN ('Red', 'Green');

-- iii. Find the Sid’s of sailors who have not reserved a boat

SELECT Sid
FROM SAILORS
WHERE Sid NOT IN (
    SELECT Sid
    FROM RESERVES
);
