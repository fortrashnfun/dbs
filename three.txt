//SQL

-- Query 1
SELECT Emp_ID
FROM EMPLOYEE
WHERE Salary = (
    SELECT MAX(Salary)
    FROM EMPLOYEE
);

-- Query 2
SELECT DISTINCT A.Aircraft_name
FROM AIRCRAFT A, CERTIFIED C, EMPLOYEE E
WHERE A.Aircraft_ID = C.Aircraft_ID
AND C.Emp_ID = E.Emp_ID
AND E.Salary > 50000;

-- Query 3
SELECT *
FROM EMPLOYEE
WHERE Emp_ID NOT IN (
    SELECT Emp_ID
    FROM CERTIFIED
);
