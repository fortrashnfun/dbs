-- Query 1
SELECT Emp_ID
FROM EMPLOYEE
WHERE Salary = (
    SELECT MAX(Salary)
    FROM EMPLOYEE
);

-- Query 2
SELECT A.Aircraft_name
FROM AIRCRAFT A
WHERE A.Aircraft_ID IN (
    SELECT C.Aircraft_ID
    FROM CERTIFIED C
    JOIN EMPLOYEE E
    ON C.Emp_ID = E.Emp_ID
    GROUP BY C.Aircraft_ID
    HAVING MIN(E.Salary) > 50000
);

-- Query 3
SELECT *
FROM EMPLOYEE
WHERE Emp_ID NOT IN (
    SELECT Emp_ID
    FROM CERTIFIED
);
